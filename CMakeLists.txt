# =============================================================================
#  CoveringArrays — a trimmed-down build of covarray-cpp
#
#  This is deliberately the smallest CMake file that still does the real job:
#  compile the covarray library, link the covarray_cli tool against it, and
#  (optionally) build the GoogleTest suite. There is no Doxygen, no Google
#  Benchmark, no install/export machinery, no CMakePresets, no sanitizers, no
#  coverage, no multi-compiler branching — GCC/g++ on Linux is the only target,
#  so every "which compiler?" branch from the original project is just gone.
#
#  Every line below does something you can point at; nothing is generated by a
#  CMake helper macro you'd have to go read elsewhere (no
#  configure_package_config_file, no write_basic_package_version_file). The one
#  CMake-generated file this project has (covarray/config.hpp) is produced by
#  the plain configure_file() call below, which just does textual @VAR@
#  substitution into a template — see include/covarray/config.hpp.in.
# =============================================================================

cmake_minimum_required(VERSION 3.20)

project(covarray LANGUAGES CXX)

# This project only ever gets built with GCC/g++, so there is no point
# carrying Clang/MSVC branches (the original project's CMake had several).
# Fail fast and clearly instead of silently compiling with the wrong flags.
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR
      "CoveringArrays only supports GCC/g++. "
      "Configure with -DCMAKE_CXX_COMPILER=g++ (and -DCMAKE_C_COMPILER=gcc).")
endif()

# C++20 is required by the library headers (std::span, etc.).
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)        # -std=c++20, never -std=gnu++20

# Feeds clangd/IDE tooling if you use it; harmless otherwise.
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# A plain `cmake -S . -B build` with no -DCMAKE_BUILD_TYPE picks a sane default
# instead of building fully unoptimized with no debug info.
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE "RelWithDebInfo" CACHE STRING "Build type" FORCE)
endif()

# -----------------------------------------------------------------------------
#  Warnings
#
#  One INTERFACE library carrying our warning flags, linked PRIVATE by every
#  target below. PRIVATE is what keeps these flags off GoogleTest (vendored
#  below) and off any future consumer that links against this project.
#  Hardcoded, not probed with check_cxx_compiler_flag(): we only ever build
#  with GCC, and every flag here is a GCC flag.
# -----------------------------------------------------------------------------
add_library(covarray_warnings INTERFACE)
target_compile_options(covarray_warnings INTERFACE
  -Wall
  -Wextra
  -Wpedantic
  -Wshadow
  -Wnon-virtual-dtor
  -Wold-style-cast
  -Wcast-align
  -Wunused
  -Woverloaded-virtual
  -Wconversion
  -Wsign-conversion
  -Wnull-dereference
  -Wdouble-promotion
  -Wformat=2
  -Wimplicit-fallthrough
  -Wmisleading-indentation
  -Wduplicated-cond
  -Wduplicated-branches
  -Wlogical-op
  -Wuseless-cast
)

# -----------------------------------------------------------------------------
#  Generated version header
#
#  configure_file() is a plain CMake builtin: it copies the .in file to the
#  destination, replacing every @VAR@ with that CMake variable's value (@ONLY
#  means "only @VAR@, ignore ${VAR}" — so the C++ preprocessor's own syntax in
#  the template is left alone). The result lands in the build tree, not in
#  include/, so it never gets confused with the hand-written headers.
# -----------------------------------------------------------------------------
set(COVARRAY_VERSION "0.1.0")

configure_file(
  "${CMAKE_CURRENT_SOURCE_DIR}/include/covarray/config.hpp.in"
  "${CMAKE_CURRENT_BINARY_DIR}/generated/include/covarray/config.hpp"
  @ONLY
)

# -----------------------------------------------------------------------------
#  The library
# -----------------------------------------------------------------------------
add_library(covarray STATIC
  src/array.cpp
  src/canonical.cpp
  src/coverage.cpp
  src/io.cpp
)

target_include_directories(covarray PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}/include"
  "${CMAKE_CURRENT_BINARY_DIR}/generated/include"
)

target_link_libraries(covarray PRIVATE covarray_warnings)

# -----------------------------------------------------------------------------
#  The command line tool
# -----------------------------------------------------------------------------
add_executable(covarray_cli tools/covarray_cli.cpp)
target_link_libraries(covarray_cli PRIVATE covarray covarray_warnings)

# -----------------------------------------------------------------------------
#  Tests (GoogleTest, vendored under third_party/googletest)
#
#  Vendored rather than fetched with FetchContent: this project is built in
#  network-restricted sandboxes, where a GitHub clone at configure time would
#  just fail. third_party/googletest/ holds the subset of the GoogleTest
#  v1.17.0 source tree needed to build the gtest/gtest_main targets (its own
#  test suite, samples, and docs are left out — nothing here uses them).
# -----------------------------------------------------------------------------
option(COVARRAY_BUILD_TESTS "Build the GoogleTest suite and register it with CTest" ON)

if(COVARRAY_BUILD_TESTS)
  enable_testing()

  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE) # we never `cmake --install` this project
  set(BUILD_GMOCK    OFF CACHE BOOL "" FORCE) # gmock is unused; skip compiling it
  add_subdirectory(third_party/googletest)

  add_executable(covarray_tests
    tests/test_array.cpp
    tests/test_canonical.cpp
    tests/test_coverage.cpp
    tests/test_io.cpp
  )
  target_link_libraries(covarray_tests PRIVATE covarray gtest_main covarray_warnings)

  # include(GoogleTest) ships with CMake; gtest_discover_tests() runs the test
  # binary's --gtest_list_tests at build/test time to register each individual
  # TEST() with CTest, instead of registering the whole binary as one test.
  include(GoogleTest)
  gtest_discover_tests(covarray_tests)
endif()
